users / companies / company_invites: equality on company_id or companies.id. Never EXISTS (SELECT 1 FROM users ...) on the users table itself.

calendar_events: user_id IS NULL OR the EXISTS predicate.
INSERT leave_requests WITH CHECK also (select auth.uid()) = user_id.
